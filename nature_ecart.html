produits = [ {"nom": "Tomates", "offre": 200, "demande": 270}, {"nom": "Pommes de terre", "offre": 1000, "demande": 100}, {"nom": "Oranges", "offre": 400, "demande": 400}, ] for p in produits: ecart = p["offre"] - p["demande"] # écart offre - demande if ecart > 0: # offre > demande print(f"ALERTE {p['nom']} : offre supérieure à la demande (+{ecart} kg)") print(" -> Baisser le prix ou lancer une promotion, réduire le prochain stock") elif ecart < 0: # demande > offre print(f"ALERTE {p['nom']} : demande supérieure à l'offre ({ecart} kg)") print(" -> Augmenter le stock (réapprovisionner) et/ou augmenter le prix") else: print(f"OK {p['nom']} : marché équilibré, aucune action")
